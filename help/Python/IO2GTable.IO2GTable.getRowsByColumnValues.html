public method O2GTable.get_rows_by_column_values

Brief

Gets rows containing any of the specified column values.

Declaration
Python
get_rows_by_column_values((str)column_name, (O2GRelationalOperators)condition, (typing.List[typing.object])column_values) -> O2GTableIterator

Parameters
column_name

The name of a column in the table.
condition

The condition the column values must satisfy.
column_values

The values of the column specified by the column_name parameter.

Details

This method allows finding all rows that contain any of the specified column values.
If no row is found, the method returns an empty iterator.

Example: Printing rows that contain any of the specified instruments (EUR/USD and GBP/USD) from the Offers table. [show]

Example: Printing rows that contain any of the specified instruments (EUR/USD and GBP/USD) from the Offers table. [hide]

from forexconnect import fxcorepy, ForexConnect
 
#Get notifications on the trading session status changes
def session_status_changed(session: fxcorepy.O2GSession,
                           status: fxcorepy.AO2GSessionStatus.O2GSessionStatus):
    print("Trading session status: " + str(status))
 
 
def main():
 
    with ForexConnect() as fx:
        try:
           #Log in
           fx.login("user_id", "password", "fxcorporate.com/Hosts.jsp",
                    "connection", session_status_callback=session_status_changed)
 
           #Get Offers table and print rows that contain specified instruments (EUR/USD and GBP/USD)
           offers_table = fx.get_table(ForexConnect.OFFERS)
           offers_rows = offers_table.get_rows_by_column_values("instrument", fxcorepy.O2GRelationalOperators.EQUAL_TO, ["EUR/USD", "GBP/USD"])
           for row in offers_rows:
               print ("Instrument: {0}, Bid: {1}, Low {2}, Ask: {3}, High {4}".format(row.instrument, row.bid, row.low, row.ask, row.high))
 
        except Exception as e:
            print("Exception: " + str(e))
 
        #Log out
        try:
            fx.logout()
        except Exception as e:
            print("Exception: " + str(e))
 
 
if __name__ == "__main__":
    main()
