from forexconnect import fxcorepy, ForexConnect
 
#Get notifications on the trading session status changes
def session_status_changed(session: fxcorepy.O2GSession,
                           status: fxcorepy.AO2GSessionStatus.O2GSessionStatus):
    print("Trading session status: " + str(status))
 
 
def main():
 
    with ForexConnect() as fx:
        try:
           #Log in
           fx.login("user_id", "password", "fxcorporate.com/Hosts.jsp",
                     "connection", session_status_callback=session_status_changed)
 
           #Get Accounts table and print all its rows
           accounts_table = fx.get_table(ForexConnect.ACCOUNTS)
           for account_row in accounts_table:
               print("AccountID: {0}, Balance: {1}, BaseUnitSize: {2}, DayPL: {3} ".format(account_row.account_id, account_row.balance,
                   account_row.base_unit_size, account_row.day_pl))
           return accounts_table.get_row(0)
 
        except Exception as e:
            print("Exception: " + str(e))
 
        #Log out
        try:
            fx.logout()
        except Exception as e:
            print("Exception: " + str(e))
 
 
if __name__ == "__main__":
    main()
