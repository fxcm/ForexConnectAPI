    O2G2Ptr<IO2GRequestFactory> factory = session->getRequestFactory();
    if (factory)
    {
       O2G2Ptr<IO2GValueMap> valuemap = factory->createValueMap();
       valuemap->setString(Command, O2G2::Commands::AcceptOrder);
       valuemap->setString(OrderID, order->getOrderID());
 
       O2G2Ptr<IO2GRequest> request = factory->createOrderRequest(valuemap);
       if (request == NULL)
           std::cout << "Order's request can not be created: " << factory->getLastError() << std::endl;
       else
           mSession->sendRequest(request);
    }
