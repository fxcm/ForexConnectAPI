public method IPriceHistoryCommunicatorListener.onRequestCompleted

Brief

The method is called when the request execution has been successfully completed.

Declaration
C#
void  onRequestCompleted (IPriceHistoryCommunicatorRequest request, IPriceHistoryCommunicatorResponse response)

Parameters
request

The request.
response

The response received from the server.

Details

To read the response content, create an instance of fxcore2.O2GMarketDataSnapshotResponseReader using the IPriceHistoryCommunicator.createResponseReader() method.

using System;
using fxcore2;
using Candleworks.PriceHistoryMgr;
...
class MyClass : IPriceHistoryCommunicatorListener
{
    ...
 
    public void processBar(DateTime date, double bidOpen, double binHigh, double bidLow, double bidClose,
                           double askOpen, double askHigh, double askLow, double askClose, int volume)
    {
       ...
    }
 
    public void processLastBar(DateTime date, int volume)
    {
        ...
    }
 
    public void onRequestCompleted(IPriceHistoryCommunicatorRequest request, IPriceHistoryCommunicatorResponse response)
    {
        try
        {
            //create a reader for the response
            O2GMarketDataSnapshotResponseReader reader = mPriceHistoryCommunicator.createResponseReader(response);
            //and read all bars
            for (int i = 0; i < reader.Count; i++)
            {
                processBar(reader.getDate(i), reader.getBidOpen(i), reader.getBidHigh(i), reader.getBidLow(i), reader.getBidClose(i),
                           reader.getAskOpen(i), reader.getAskHigh(i), reader.getAskLow(i), reader.getAskClose(i),
                           reader.getVolume(i));
            }
            //finally notify the collection that all bars are added, so it can
            //add all ticks collected while the request was being executed
            //and start update the data by forthcoming ticks
            processLastBar(reader.getLastBarTime(), reader.getLastBarVolume());
        }
        catch (Exception e)
        {
            //process error
 
        }
    }
 
}
