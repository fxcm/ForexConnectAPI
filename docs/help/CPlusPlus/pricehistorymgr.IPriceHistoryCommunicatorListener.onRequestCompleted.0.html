public method IPriceHistoryCommunicatorListener::onRequestCompleted

Brief

The method is called when the request execution has been successfully completed.

Declaration
C++
virtual void  onRequestCompleted (IPriceHistoryCommunicatorRequest *request, IPriceHistoryCommunicatorResponse *response) = 0

Parameters
request

The request.
response

The response received from the server.

Details

To read the response content, create an instance of IO2GMarketDataSnapshotResponseReader using the IPriceHistoryCommunicator::createResponseReader() method.

Example [show]

Example [hide]

#include "ForexConnect.h"
#include "pricehistorymgr/pricehistorymgr.h"
 
using namespace pricehistorymgr;
 
class MyClass : public IPriceHistoryCommunicatorListener
{
    ...
    IPriceHistoryCommunicator *mPriceHistoryCommunicator;
 
    ...
    void onRequestCompleted(IPriceHistoryCommunicatorRequest *request,
                            IPriceHistoryCommunicatorResponse *response)
    {
        //create a reader for the response
        IError *error;
        IO2GMarketDataSnapshotResponseReader *reader =
            mPriceHistoryCommunicator->createResponseReader(response, &error);
 
        if (reader == NULL)
        {
            if (error != NULL)
            {
                //process error
                //...
                error->release();
            }
            return;
        }
 
        //and read all bars
        for (int i = 0; i < reader->size(); ++i)
        {
            processBar(reader->getDate(i), reader->getBidOpen(i), reader->getBidHigh(i),
                reader->getBidLow(i), reader->getBidClose(i), reader->getAskOpen(i),
                reader->getAskHigh(i), reader->getAskLow(i), reader->getAskClose(i),
                reader->getVolume(i));
        }
 
        //finally notify the collection that all bars are added, so it can
        //add all ticks collected while the request was being executed
        //and start update the data by forthcoming ticks
        processLastBar(reader->getLastBarTime(), reader->getLastBarVolume());
        reader->release();
    }
};
