#include <windows.h>

 

class MyRefCountClass : public IAddRef

{

 private:

    /** Reference counter. */

    long mRefCount;

 protected:

    /** Destructor. */

    virtual ~MyRefCountClass();

 public:

    /** Constructor. */

    MyRefCountClass();

 

    /** Increase reference counter. */

    virtual long addRef();

 

    /** Decrease reference counter. */

    virtual long release();

};

 

MyRefCountClass::MyRefCountClass()

{

    mRefCount = 1;

}

 

MyRefCountClass::~MyRefCountClass()

{

}

 

/** Increase reference counter. */

long MyRefCountClass::addRef()

{

    return InterlockedIncrement(&mRefCount);

}

 

/** Decrease reference counter. */

long MyRefCountClass::release()

{

    long rc = InterlockedDecrement(&mRefCount);

    if (rc == 0)

        delete this;

    return rc;

}
